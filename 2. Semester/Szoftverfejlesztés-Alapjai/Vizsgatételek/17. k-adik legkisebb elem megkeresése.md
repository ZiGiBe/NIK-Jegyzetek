```
függvény KLegkisebb(x, bal, jobb, k)
	ha bal = jobb akkor
		vissza x[bal]
	különben
		idx <- Szétválogat(x, bal, jobb)
		ha k = idx - bal + 1 akkor
			vissza x[idx]
		különben ha k < idx - bal + 1 akkor
			KLegkisebb(x, bal, idx - 1, k)
		különben
			KLegkisebb(x, idx + 1, jobb, k - (idx - bal + 1))
		elágazás vége
	elágazás vége
függvény vége

függvény Szétválogat(x, bal, jobb)
	segéd <- x[bal]
	ciklus amíg (bal <= jobb)
		ciklus amíg (bal < jobb) és (x[jobb] > segéd)
			jobb <- jobb - 1
		ciklus vége
		ha bal < jobb
			x[bal] <- x[jobb]
			bal <- bal + 1
		elágazás vége
		ciklus amíg (bal < jobb) és (x[bal] <= segéd)
			x[jobb] <- x[bal]
			jobb <- jobb - 1
		elágazás vége
	ciklus vége
	idx <- bal
	x[idx] <- segéd
	vissza idx
függvény vége
```
- Futási idő: 
	- Átlagosan: O(2n-1) = O(n)
	- Legrosszabb esetben: O(n^2) (Rendezett tömbök esetén)
