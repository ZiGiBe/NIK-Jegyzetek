## Rendezettség
```
függvény Rendezett(x, n)
	i <- 1
	ciklus amíg (i<=i-1) és (x[i] <= x[i+1])
		i<-i+1
	ciklus vége
	out <- i >= n
	vissza out
függvény vége
```
## Bináris keresés (iteratív)
```
függvény BinárisKeresés(x, n, k)
	bal <- 1
	jobb <- n
	center <- (bal+jobb)/2
	ciklus amíg (bal <= jobb) és (x[center]!=k)
		ha x[center] < k akkor
			bal <- center
		különben ha x[center] > k akkor
			jobb <- center - 1
		elágazás vége
		center <- (bal+jobb)/2
	ciklus vége
	van <- bal<=jobb
	ha van akkor
		idx <- center
		vissza(van, idx)
	különben
		vissza van
	elágazás vége
függvény vége
```
- Futási idő: 
	- Legjobb esetben: O(1)
	- Legrosszabb esetben: O(log(n))
## Bináris keresés (rekurzív)
```
függvény BinárisKeresésRekurzív(x, bal, jobb, k)
	ha bal > jobb akkor
		vissza 0
	különben
		center <- (bal + jobb) / 2
		ha (x[center]=k) akkor
			vissza center
		különben ha x[center] < k akkor
			BinárisKeresésRekurzív(x, center - 1, jobb, k)
		különben
			BinárisKeresésRekurzív(x, bal, center + 1, k)
		elágazás vége
	elágazás vége
függvény vége
```
## Alsó határ
```
függvény FelsőHatár(x, n, k)
	bal <- 1
	jobb <- n
	idx <- 0
	center <- (bal + jobb) / 2
	ciklus amíg bal <= jobb
		ha x[center] < k akkor
			idx <- center
			bal <- center + 1
		különben
			jobb <- center - 1
		elágazás vége
	ciklus vége
	vissza idx
függvény vége
```
