```
eljárás QuickSort(cím x, bal, jobb)
	idx <- Szétválogat(x, bal, jobb)
	ha idx > bal + 1 akkor
		QuickSort(x, bal, idx - 1)
	elágazás vége
	ha idx < jobb - 1 akkor
		QuickSort(x, idx + 1, jobb)
	elágazás vége
eljárás vége

függvény Szétválogat(cím x, bal, jobb)
	segéd <- x[bal]
	ciklus amíg bal < jobb
		ciklus amíg bal < jobb és x[jobb] > segéd
			jobb <- jobb - 1
		ciklus vége
		ha bal < jobb akkor
			x[bal] <- x[jobb]
			bal <- bal + 1
		elágazás vége
		ciklus amíg bal < jobb és x[bal] <= segéd
			bal <- bal + 1
		ciklus vége
		ha bal < jobb akkor
			x[jobb] <- x[bal]
			jobb <- jobb - 1
		elágazás vége
	ciklus vége

	idx <- bal
	x[idx] <- segéd
	vissza idx
függvény vége
```
- Futási idő: O(n*log(n))
	- Meghívások száma: átlagosan n*log(n)
	- Legrosszabb eset, ha a tömb rendezett, vagy fordítva rendezett, ilyenkor O(n^2)
